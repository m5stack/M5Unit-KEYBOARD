617 bool permitted_mode(const uint8_t mbits) const
618 {
619 uint8_t mod8 = static_cast<uint8_t>(_state.now.to_ullong() >> 56);
620 // mod8 bit0=Shift→mbits bit1, bit1=Sym→bit2, bit2=Fn→bit3
621 return mbits & (mod8 ? (mod8 << 1) : 0x01);
622 }
